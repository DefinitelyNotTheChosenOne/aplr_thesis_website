            positions[i3 + 1] = (Math.random() - 0.5) * 100;

                colors[i3] = 0.133; colors[i3 + 1] = 0.773; colors[i3 + 2] = 0.369; // green

            sizes[i] = Math.random() * 2 + 0.5;

        particlesGeometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));

        const particlesMaterial = new THREE.PointsMaterial({

            opacity: 0.6,
            blending: THREE.AdditiveBlending,
            sizeAttenuation: true,
        });

        const particles = new THREE.Points(particlesGeometry, particlesMaterial);
        scene.add(particles);

        // Floating geometric shapes

        // Icosahedrons

        function animate() {
            requestAnimationFrame(animate);

            const time = Date.now() * 0.001;

            // Smooth scroll interpolation
            currentScrollY += (scrollY - currentScrollY) * 0.1;
            mouseX += (targetMouseX - mouseX) * 0.05;
            mouseY += (targetMouseY - mouseY) * 0.05;

            // Rotate particles based on scroll
            const scrollFactor = currentScrollY * 0.0003;
            particles.rotation.y = scrollFactor + time * 0.02;

            camera.position.x += (mouseX * 3 - camera.position.x) * 0.02;
            camera.position.y += (-mouseY * 3 - camera.position.y) * 0.02;

            particles.position.y = -currentScrollY * 0.01;

            for (let i = 0; i < particlesCount; i++) {

                shape.rotation.z += shape.userData.rotSpeed.z;
                shape.position.y = shape.userData.originalY + 
                    Math.sin(time * shape.userData.floatSpeed + shape.userData.floatOffset) * 2;
                // Scroll parallax on shapes
                shape.position.y -= currentScrollY * 0.005;
            });

            // Color shift based on scroll section
            const totalHeight = document.body.scrollHeight - window.innerHeight;
            const scrollProgress = currentScrollY / totalHeight;

            const colArray = particlesGeometry.attributes.color.array;
            for (let i = 0; i < particlesCount; i++) {
                const i3 = i * 3;
                if (scrollProgress < 0.25) {
                    // Indigo / purple
                } else if (scrollProgress < 0.5) {

                    colArray[i3] += (0.2 - colArray[i3]) * 0.01 * t;
                    colArray[i3 + 2] += (0.95 - colArray[i3 + 2]) * 0.01 * t;
                } else if (scrollProgress < 0.75) {
                    const t = (scrollProgress - 0.5) / 0.25;
                    if (i % 3 === 0) {
                        colArray[i3] += (0.9 - colArray[i3]) * 0.005 * t;
                        colArray[i3 + 1] += (0.5 - colArray[i3 + 1]) * 0.005 * t;
                        colArray[i3 + 2] += (0.1 - colArray[i3 + 2]) * 0.005 * t;
                    }
                }
            }
            particlesGeometry.attributes.color.needsUpdate = true;

            renderer.render(scene, camera);
